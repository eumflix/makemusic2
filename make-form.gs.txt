/* 나도 작곡가 평가 — 구글폼 자동 만들기
 * 구글 스프레드시트 → 확장 프로그램 → Apps Script 에 붙여 넣고, 「평가폼만들기」를 한 번 실행하세요.
 * 폼을 만들고, 응답이 이 스프레드시트에 쌓이도록 연결하고, 학생용 주소를 '학생용 주소' 시트에 적어 줘요. */
const SITE = 'https://eumflix.github.io/makemusic2/eval.html';

// [문항 제목, 앱이 알아보는 낱말, 긴 글 여부]
const FIELDS = [
  ['이름', '이름', false], ['반', '반', false], ['레벨', '레벨', false], ['마디', '마디', false], ['총점', '총점', false],
  ['코드 정답률', '코드', false], ['음 정답률', '음', false], ['리듬 정답률', '리듬', false], ['화면 이탈', '이탈', false],
  ['총평', '총평', true], ['결과 전문', '결과', true], ['제출시각', '시각', false]
];
const KEYS = { '이름': 'name', '반': 'klass', '레벨': 'level', '마디': 'bars', '총점': 'score', '코드': 'accChord', '음': 'accNote', '리듬': 'accRhythm', '이탈': 'away', '총평': 'comment', '결과': 'text', '시각': 'at' };

function onOpen() {
  try { SpreadsheetApp.getUi().createMenu('나도 작곡가').addItem('평가 폼 만들기', '평가폼만들기').addToUi(); } catch (e) {}
}

function 평가폼만들기() {
  const form = FormApp.create('나도 작곡가 평가 결과');
  form.setDescription('학생이 평가를 마치면 결과가 자동으로 이곳에 쌓여요. 이 폼 주소는 학생에게 알려 주지 않아도 돼요.');
  try { form.setCollectEmail(false); } catch (e) {}
  try { form.setLimitOneResponsePerUser(false); } catch (e) {}
  try { form.setRequireLogin(false); } catch (e) {}
  try { form.setConfirmationMessage('제출되었습니다.'); } catch (e) {}

  const ss = SpreadsheetApp.getActiveSpreadsheet() || SpreadsheetApp.create('나도 작곡가 평가 결과');
  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());      // 응답이 이 스프레드시트에 쌓여요

  const pre = form.createResponse();                                          // 「양식 미리 작성」과 같은 방식으로 문항 번호를 알아내요
  FIELDS.forEach(f => {
    const it = f[2] ? form.addParagraphTextItem() : form.addTextItem();
    it.setTitle(f[0]);
    pre.withItemResponse(f[2] ? it.asParagraphTextItem().createResponse(f[1]) : it.asTextItem().createResponse(f[1]));
  });

  const link = pre.toPrefilledUrl(), id = (link.match(/\/forms\/d\/e\/([^\/]+)/) || [])[1], map = {};
  const q = link.split('?')[1] || '';
  q.split('&').forEach(p => { const m = p.match(/^entry\.(\d+)=(.*)$/); if (m) { const k = KEYS[decodeURIComponent(m[2].replace(/\+/g, ' '))]; if (k) map[m[1]] = k; } });
  if (!id || !Object.keys(map).length) throw new Error('폼 주소를 읽지 못했어요. 다시 실행해 주세요.');

  const code = Utilities.base64EncodeWebSafe(JSON.stringify({ i: id, m: map })).replace(/=+$/, '');
  const url = SITE + '?f=' + code;

  const sh = ss.getSheetByName('학생용 주소') || ss.insertSheet('학생용 주소');
  sh.clear(); sh.getRange('A1').setValue('학생에게 나눠 줄 주소 (평가 화면)'); sh.getRange('A2').setValue(url);
  sh.getRange('A4').setValue('응답 폼(선생님용)'); sh.getRange('A5').setValue(form.getEditUrl());
  sh.setColumnWidth(1, 640);
  try { SpreadsheetApp.getUi().alert('완료!\n\n「학생용 주소」 시트의 A2 칸 주소를 학생들에게 나눠 주세요.\n제출된 결과는 이 스프레드시트의 「양식 응답」 시트에 쌓여요.'); } catch (e) {}
  return url;
}
